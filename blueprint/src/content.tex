\chapter{Preliminaries}

This blueprint follows Fan Chung: Spectral Graph Theory (first published: AMS, 1992).\\
Unless otherwise specified, all matrices are $n \times n$ where $n$ is the number of vertices in $G$.
We omit natural language proofs for theorems that only involve matrix multiplication and basic operations, although we formally prove them in Lean.

\section{Weighted Graph}

\begin{definition}[Weighted Graph]
\label{def:UnweightedGraph}
\lean{UnweightedGraph}
\leanok
We extend the definition of a general graph to that of a weighted graph by defining a weight function over its edges.

\[
\omega : \beta \to \mathbf{R}_{+}
\]

with the condition that

\[
e \in edgeSet \iff 0 < w(e)
\]

We also add an arbitray orientation for undirected edges.

\end{definition}


\begin{definition}
\label{def:degree}    
\lean{UnweightedGraph.degree}
\leanok
In a graph G, let $d_v$ denote the degree (number of incident edges) of vertex $v$.
\end{definition}



\begin{definition}
\lean{UnweightedGraph.IsRegularOfDegree}
\uses{def:degree}
\label{def:IsRegularOfDegree}
\leanok
For $k \in \mathbb{N}$, a graph is said to be $k$-regular if every vertex $v$ has degree $d_v = k$.
\end{definition}


\begin{definition}
\lean{UnweightedGraph.graphVolume}
\label{def:graphVolume}
\uses{def:degree}
\leanok
The volume of a graph is defined as the sum of the degrees of its vertices.
\end{definition}



\begin{definition}
\lean{UnweightedGraph.IsIsolated}
\label{def:IsIsolated}
\uses{def:degree}
\leanok
A vertex v is said to be isolated if $d_v = 0$.
\end{definition}



\begin{definition}
\lean{UnweightedGraph.NonTrivial}
\label{def:NonTrivial}
\leanok
A graph is said to be nontrivial if it contains at least one edge.
\end{definition}



\section{Matrices and Operators}

\begin{definition}
\lean{UnweightedGraph.Adjacency}
\label{def:Adjacency}
\leanok
The adjacency matrix of a graph $G$, denoted $A$ is defined as follows,
\[
A(u,v)=
\begin{cases}
    1 & \text{if } u \text{ and } v \text{ are adjacent}, \\
    0 & \text{otherwise}.
\end{cases}
\]
\end{definition}


\begin{definition}
\lean{UnweightedGraph.L}
\label{def:L}
\leanok
Consider the matrix L, defined as follows
\[
L(u,v)=
\begin{cases}
  d_v & \text{if}\ u=v, \\
  -1 & \text{if}\ u \text{ and }v \text{ are ajacent},\\
    0 & \text{otherwise}.
\end{cases}
\]
\end{definition}



\begin{definition} [Laplacian Matrix]
\lean{UnweightedGraph.Laplacian}
\label{def:Laplacian}
\leanok
Consider the matrix $\mathcal{L}$, defined as follows
\[
\mathcal{L}(u,v) = \left\{
\begin{array}{cl}
    1 & \text{if } u = v \text{ and } d_v \neq 0, \\[1.5ex]
    -\dfrac{1}{\sqrt{d_u d_v}} & \text{if } u \text{ and } v \text{ are adjacent}, \\[1.5ex]
    0 & \text{otherwise}.
\end{array}
\right.
\]
\end{definition}




\begin{definition}
\lean{UnweightedGraph.T_sqrt}
\label{def:T_sqrt}
\leanok
Let $T$ denote the diagonal matrix with the $(u,v)$-th entry having value $d_v$.
\end{definition}



\begin{definition}
\lean{UnweightedGraph.T_inv_sqrt}
\label{def:T_inv_sqrt}
\leanok
We also define $T^{-1/2}$ with the convention that $T^{-1/2} = 0$ when $d_v = 0$.
\end{definition}



\begin{lemma} [Laplacian symmetric normalization]
\lean{UnweightedGraph.Lap_symmetric_normalization}
\label{lem:Lap_symmetric_normalization}
\uses{def:T_inv_sqrt, def:Laplacian, def:L}
\leanok
\[
\mathcal{L} = T^{-1/2} \text{ } L \text{ } T^{-1/2}
\]
\end{lemma}



\begin{definition}
\lean{UnweightedGraph.LapOperator}
\label{def:LapOperator}
\leanok
The Laplacian of a graph can be viewed as an operator on the space of functions $g : V(G) \rightarrow \mathbb{R}$.
\end{definition}


\begin{definition}
\lean{UnweightedGraph.LapOperatorFormula}
\label{def:LapOpFormula}
\uses{def:LapOperator}
\leanok
The Laplacian Operator satisfies
\[
\mathcal{L}g(u) = \frac{1}{\sqrt{d_u}} \sum_{\substack{v \\ u \sim v}} \left( \frac{g(u)}{\sqrt{d_u}} - \frac{g(v)}{\sqrt{d_v}} \right).
\]
\end{definition}



\begin{lemma}
\lean{UnweightedGraph.LapOfRegGraph}
\label{lem:LapOfRegGraph}
\uses{def:Laplacian, def:Adjacency, def:IsRegularOfDegree}
\leanok
When $G$ is $k$-regular,
\[
\mathcal{L} = I - \frac{1}{k} A,
\]  
\end{lemma}
where $k > 0$, $A$ is the adjacency matrix of $G$, and $I$ is the identity matrix.



\begin{lemma}
\lean{UnweightedGraph.LapOfNotIsolatedGraph}
\label{lem:LapOfNotIsolatedGraph}
\uses{def:IsIsolated}
\leanok
For a general graph without isolated vertices, we have
\[
\begin{aligned}
\mathcal{L} = I - T^{-1/2} A T^{-1/2}.
\end{aligned}
\]
\end{lemma}


\begin{definition}[Boundary Operator]
\lean{UnweightedGraph.S}
\label{def:S}
\uses{def:degree}
\leanok
Let $S$ denote the matrix whose rows are indexed by the vertices and whose columns
are indexed by the edges of G. Each column corresponding to an edge $e = \{u, v\}$
has an entry $1/\sqrt{d_u}$ in the row corresponding to $u$, an entry $−1/\sqrt{d_v}$ in
the row corresponding to $v$, and has zero entries elsewhere.
\end{definition}



\begin{lemma}
\label{lem:LSS}
\lean{UnweightedGraph.LSS}
\uses{def:Laplacian, def:S}
We note that $\mathcal{L}$ can be written as
\[
\mathcal{L} = SS^*.
\]
\end{lemma}



\begin{lemma}
\lean{UnweightedGraph.LapHermitian}
\label{lem:LapHermitian}
\uses{def:Laplacian}
\leanok
The Matrix $\mathcal{L}$ is hermitian.
\end{lemma}

\begin{proof}
By spectral theorem.
\end{proof}



\begin{definition}
\lean{UnweightedGraph.lapEigvals}
\label{def:lapEigvals}
\uses{lem:LapHermitian}
\leanok
Following \ref{lem:LapHermitian}, the eigenvalues of $\mathcal{L}$ are all real and non-negative.
\end{definition}




\begin{definition}
\lean{UnweightedGraph.DirichletSum}
\label{def:DirichletSum}
\leanok
The Dirichlet sum of a graph $G$ for a function $f : \alpha \to \mathbb{R}$ is defined as
\[
\sum_{u \sim v}{(f(u)-f(v))^2}.
\]
\end{definition}




\begin{definition}
Let $\tau$ denote the constant function which assigns the value $1$ on each vertex.
\end{definition}


\begin{theorem}
\lean{UnweightedGraph.zero_mem_lapSpectrum}
\label{thm:zero_mem_lapSpectrum}
\uses{def:T_sqrt, def:Laplacian}
\leanok
$T^{1/2} * \tau$ is an eigenfunction of $\mathcal{L}$ with eigenvalue $0$.
\end{theorem}


\section{Basic facts about the spectrum of a graph}

\begin{lemma}
\lean{UnweightedGraph.eigval_sum_le_n}
\label{lem:eigval_sum_le_n}
\uses{def:lapEigvals}
\leanok
The sum of the eigenvalues of the graph is at most its number of vertices, that is
\[
\sum_{i}{\lambda_i} \le n.
\]
\end{lemma}

\begin{proof}
Follows from considering the trace of $\mathcal{L}$.
\end{proof}



\begin{lemma}
\lean{UnweightedGraph.eigval_sum_eq_n_iff_no_isolation}
\label{lem:eigval_sum_eq_n_iff_no_isolation}
\uses{def:lapEigvals}
\leanok
In the previous lemma, the equality holds if and only if $G$ has no isolated vertices.
\end{lemma}



\begin{lemma}
\lean{UnweightedGraph.second_eigval_le_div}
\label{lem:second_eigval_le_div}
\uses{lem:eigval_sum_le_n}
When $n>1$, we obtain the following upper bound on the second eigenvalue:
\[
\lambda_1 \le \frac{n}{n-1}.
\]
\end{lemma}




\chapter{Connectivity of general graph}

\begin{definition}[Walk]
\label{def:Walk}
\lean{UnweightedGraph.Walk}
\leanok
A walk is a sequence of adjacent vertices. For vertices $u, v \in V$, the walk between them is a sequence starting at $u$ and ending at $v$.
\end{definition}

\begin{definition}[Reachable]
\label{def:Reachable}
\lean{UnweightedGraph.Reachable}
\uses{def:Walk}
\leanok
Two vertices $u$ and $v$ are said to be reachable if there is a walk between them.
\end{definition}

\begin{definition}[Preconnected]
\label{def:Preconnected}
\lean{UnweightedGraph.Preconnected}
\uses{def:Reachable}
\leanok
A graph is preconnected if every pair of vertices is reachable from one another.
\end{definition}

\begin{definition}[Connected]
\label{def:Connected}
\lean{UnweightedGraph.Connected}
\uses{def:Preconnected}
\leanok
A graph is connected if it is preconnected and contains at least one vertex.
\end{definition}

\begin{definition}[Edge Boundary]
\label{def:edgeBoundary}
\lean{UnweightedGraph.edgeBoundary}
\leanok
For a subset of vertices $S \subseteq V$, the edge boundary $\partial S$ consists of all edges with exactly one endpoint in $S$ and the other in $S^c$.
\end{definition}

\begin{lemma}
\label{lem:walk_crossing}
\lean{UnweightedGraph.walk_crossing}
\uses{def:Walk, def:edgeBoundary}
If there is a walk going from a vertex $u \in S$ to a vertex $v \in S^c$, then there is at least one edge that goes from $S$ to $S^c$.
\end{lemma}

\begin{lemma}
\label{lem:connected_non_empty_edge_boundary}
\lean{UnweightedGraph.connected_non_empty_edge_boundary}
\uses{def:Connected, def:edgeBoundary}
\leanok
In a connected graph, every set of vertices $S$ different from the empty set and the universal set ($S \neq \emptyset$ and $S \neq V$) has a non-empty edge boundary, that is $\partial S \neq \emptyset$.
\end{lemma}








\chapter{Isoperimetric problems}


\begin{definition}[Edge Connection]
\label{def:edgeConnection}
\lean{UnweightedGraph.edgeConnection}
\leanok
For two sets of vertices $A$ and $B$, $E(A, B)$ denotes the set of edges with one endpoint in $A$ and one endpoint in $B$.
\end{definition}

\begin{lemma}
\label{lem:edgeBoundaryComplement}
\lean{UnweightedGraph.Isoperimetry.edgeBoundaryComplement}
\uses{def:edgeBoundary}
\leanok
For any set of vertices $S$, the edge boundary of $S$ is equal to the edge boundary of its complement:
$$
\partial S = \partial S^c.
$$
\end{lemma}

\begin{lemma}
\label{lem:self_connection_eq_boundary}
\lean{UnweightedGraph.Isoperimetry.self_connection_eq_boundary}
\uses{def:edgeBoundary, def:edgeConnection}
\leanok
The edge boundary of a set $S$ is equal to the edge connection between $S$ and $S^c$:
$$
\partial S = E(S, S^c).
$$
\end{lemma}

\begin{definition}
\label{def:hG}
\lean{UnweightedGraph.Isoperimetry.hG}
\uses{def:edgeBoundary, def:graphVolume}
\leanok
For a vertex set $S$, we define $h_G(S)$ as the ratio of the size of its edge boundary to the minimum volume of $S$ and its complement:
$$
h_G(S) = \frac{|E(S, S^c)|}{\min(\operatorname{vol}(S), \operatorname{vol}(S^c))}.
$$
\end{definition}

\begin{definition}[Cheeger Constant]
\label{def:cheeger}
\lean{UnweightedGraph.Isoperimetry.cheeger}
\uses{def:hG}
\leanok
The Cheeger constant of a graph $G$, denoted $h_G$, is defined as the minimum of $h_G(S)$ for every set of vertices $S$ with non-zero volume and co-volume:
$$
h_G = \inf_{\substack{S \subset V \\ 0 < \min(\operatorname{vol}(S), \operatorname{vol}(S^c))}} h_G(S).
$$
\end{definition}

\begin{lemma}
\label{lem:cheeger_mul_volume_le_volume_frontier}
\lean{UnweightedGraph.Isoperimetry.cheeger_mul_volume_le_volume_frontier}
\uses{def:cheeger, def:edgeBoundary}
\leanok
For any set of vertices $S$ satisfying $\operatorname{vol}(S) \le \operatorname{vol}(S^c)$ and $0 < \min(\operatorname{vol}(S), \operatorname{vol}(S^c))$, we have:
$$
h_G \cdot \operatorname{vol}(S) \le |\partial S|.
$$
\end{lemma}

\begin{lemma}
\label{lem:connected_iff_cheeger_pos}
\lean{UnweightedGraph.Isoperimetry.connected_iff_cheeger_pos}
\uses{def:Connected, def:cheeger}
A graph $G$ is connected if and only if its Cheeger constant is strictly positive:
$$
G \text{ is connected} \iff 0 < h_G.
$$
\end{lemma}

\begin{lemma}
\label{lem:two_cheegerV_ge_first_eigval}
\lean{UnweightedGraph.Isoperimetry.two_cheegerV_ge_first_eigval}
\uses{def:cheeger}
We derive a simple upper bound for the first non-trivial eigenvalue $\lambda_1$ in terms of the Cheeger constant of a connected graph:
$$
\lambda_1 \le 2 h_G.
$$
\end{lemma}

\chapter{Expander graphs and the mixing lemma}

This chapter follows Swastik Kopparty, \emph{Lecture 2: Expander Graphs, Mixing lemma and
Applications to randomness}, Topics in Pseudo-randomness and Complexity Theory (Spring 2018),
Rutgers University; scribed by Danny Scheinerman and Harsha Tirumala.

Throughout, $G$ is a $d$-regular simple graph on $n$ vertices.

\begin{definition}[Normalized adjacency matrix]
\label{def:norm_adj_matrix}
\lean{norm_adj_matrix}
\leanok
The normalized adjacency matrix of $G$ is $M = \frac{1}{d} A$, where $A$ is the adjacency matrix.
Since $G$ is $d$-regular, $M$ is the transition matrix of the simple random walk on $G$.
\end{definition}

\begin{lemma}
\label{lem:norm_adj_matrix_isHermitian}
\lean{norm_adj_matrix_isHermitian}
\uses{def:norm_adj_matrix}
\leanok
$M$ is symmetric, hence Hermitian, so the spectral theorem provides an orthonormal basis of
eigenvectors $v_1, \dots, v_n$ with real eigenvalues $\lambda_1, \dots, \lambda_n$.
\end{lemma}

\begin{lemma}
\label{lem:norm_adj_matrix_mulVec_const}
\lean{norm_adj_matrix_mulVec_const}
\uses{def:norm_adj_matrix}
\leanok
Every constant function is an eigenvector of $M$ with eigenvalue $1$: each vertex has exactly
$d$ neighbours, so averaging a constant over them returns the same constant.
\end{lemma}

\begin{theorem}
\label{thm:eq_const_of_mulVec_eq}
\lean{eq_const_of_mulVec_eq}
\uses{def:norm_adj_matrix, def:Connected}
\leanok
Let $G$ be connected and $d$-regular. Then every eigenvector of $M$ with eigenvalue $1$ is
constant.
\end{theorem}

\begin{proof}
\leanok
Let $f$ satisfy $Mf = f$ and let $m = f(v_0)$ be the maximum value of $f$, attained at some
vertex $v_0$ since $V$ is finite and nonempty. Because $G$ is $d$-regular,
$$ f(v_0) = (Mf)(v_0) = \frac{1}{d} \sum_{u \sim v_0} f(u), $$
so $\sum_{u \sim v_0} (m - f(u)) = d\,m - d\,f(v_0) = 0$. Each summand is nonnegative, hence every
summand vanishes and $f(u) = m$ for every neighbour $u$ of $v_0$. Propagating this along a walk,
which exists between any two vertices since $G$ is connected, gives $f \equiv m$.
\end{proof}

\begin{lemma}
\label{lem:eigval_eq_one_unique}
\lean{eigval_eq_one_unique}
\uses{thm:eq_const_of_mulVec_eq}
\leanok
On a connected $d$-regular graph the eigenvalue $1$ is simple: two distinct basis indices cannot
both carry the eigenvalue $1$, since their eigenvectors would both be nonzero constants and so
could not be orthogonal.
\end{lemma}

\begin{definition}[Expander]
\label{def:expander}
\lean{expander}
\uses{def:norm_adj_matrix}
\leanok
$G$ is a $\gamma$-expander when every eigenvalue of $M$ other than the top eigenvalue $1$ is at
most $\gamma$ in absolute value. By \ref{lem:eigval_eq_one_unique} this excludes exactly one
eigenvalue when $G$ is connected.
\end{definition}

\begin{definition}
\label{def:e_AB}
\lean{e_AB}
\leanok
For vertex sets $A, B \subseteq V$, $e(A, B) = |\{(a,b) \in E(G) \;:\; a \in A, b \in B\}|$ is the
number of edges joining a vertex in $A$ to a vertex in $B$.
\end{definition}

\begin{theorem}[Expander mixing lemma]
\label{thm:expander_mixing_lemma}
\lean{expander_mixing_lemma}
\uses{def:expander, def:e_AB, lem:eigval_eq_one_unique}
\leanok
Let $G$ be a connected $d$-regular $\gamma$-expander on $n$ vertices and let $A, B \subseteq V$ be
two sets of vertices (possibly overlapping). Then
$$ \left| e(A,B) - \frac{d}{n} |A| |B| \right| \le \gamma\, d \sqrt{|A| |B|}. $$
Note that the estimate $e(A,B) \approx \frac{d}{n}|A||B|$ is what one would expect for a random
$d$-regular graph on $n$ vertices.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:norm_adj_matrix_isHermitian, lem:norm_adj_matrix_mulVec_const}
Let $\mathbf{1}_A, \mathbf{1}_B : V \to \mathbf{R}$ be the indicator functions of $A$ and $B$, and
write their expansions in the orthonormal eigenbasis as $\alpha_i = \langle v_i, \mathbf{1}_A
\rangle$ and $\beta_i = \langle v_i, \mathbf{1}_B \rangle$. Since $G$ is $d$-regular,
$$ e(A,B) = \sum_{i,j} \mathbf{1}_A(i) \mathbf{1}_B(j) A_{ij}
          = d \left\langle \mathbf{1}_A, M \mathbf{1}_B \right\rangle
          = d \sum_i \lambda_i \alpha_i \beta_i . $$
Let $i_0$ be the index carrying the eigenvalue $1$, which is unique by
\ref{lem:eigval_eq_one_unique}; its eigenvector is a constant $c$ with $n c^2 = 1$. Hence
$\alpha_{i_0} = c |A|$ and $\beta_{i_0} = c |B|$, so the $i_0$ term contributes
$$ d\,\lambda_{i_0} \alpha_{i_0} \beta_{i_0} = d\,c^2 |A| |B| = \frac{d}{n} |A| |B|, $$
which is exactly the main term. For the remaining terms, $|\lambda_i| \le \gamma$ and
Cauchy-Schwarz followed by Parseval give
$$ \left| \sum_{i \ne i_0} \lambda_i \alpha_i \beta_i \right|
   \le \gamma \sum_{i \ne i_0} |\alpha_i| |\beta_i|
   \le \gamma \Big( \sum_i \alpha_i^2 \Big)^{1/2} \Big( \sum_i \beta_i^2 \Big)^{1/2}
   = \gamma \sqrt{|A|}\sqrt{|B|} . $$
Multiplying by $d$ gives the claim.
\end{proof}
