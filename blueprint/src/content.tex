A graph is $G = (V, E)$ is [\dots]
Throughout this report, $\alpha$ denotes the universe in which the vertices live.


\begin{definition}
    \label{def:finitegraph}
    \lean{Graph.IsFinite}
    \leanok
    A finite graph is a graph in which the vertex set and the edge set are finite sets.
\end{definition}


\begin{definition}
    \label{def:graphwalk}
    \lean{Graph.Walk}
    \leanok
    A walk is a sequence of adjacent vertices.\\For vertices u v $\in$ $V ⊆ \alpha$, walk u v consists of all walks starting at u and ending at v.
\end{definition}


\begin{definition}
    \label{def:reachable}
    \lean{Graph.Reachable}
    \uses{def:graphwalk}
    \leanok
    Two vertices are reachable if there is a walk (a path) between them.
\end{definition}


\begin{definition}
    \label{def:preconnected}
    \lean{Graph.Preconnected}
    \uses{def:reachable}
    \leanok
    A graph is preconnected if every pair of vertices is reachable from one another.
\end{definition}


\begin{definition}
    \label{def:connected}
    \lean{Graph.Connected}
    \uses{def:preconnected}
    \leanok
    A graph is connected if it is preconnected and contains at least one vertex
\end{definition}

\section{Expansion}

\begin{definition}
    \label{def:edgeBoundary}
    \lean{ExpanderGraphs.edgeBoundary}
    \leanok
    The \textbf{Edge Boundary} of a set $S$, denoted $\partial S$, is $\partial S = E(S, \overline{S})$. This is the set of edges emanating from the set $S$ to its complement.
\end{definition}


\begin{definition}
  \label{def:cheegerConstant}
  \lean{ExpanderGraphs.cheegerConstant}
  \uses{def:edgeBoundary}
  \leanok
  The \textbf{Cheeger constant} (also called edge expansion, or conductance) of a finite graph $G$ with vertex set V, denoted $h(G)$, is defined by
  \[
  h(G) := \min \left\{ \frac{|\partial S|}{|S|} \ \middle|\ S \subseteq V, \ 0 < |S| \le \frac{|V|}{2} \right\}.
  \]
\end{definition}


\begin{lemma}
    \label{lem:connected_edgeBoundary_nonempty}
    \lean{ExpanderGraphs.connected_edgeBoundary_nonempty}
    \uses{def:edgeBoundary, def:connected}
    If G is a connected graph, every non empty proper subset of its vertices has a non empty edge boundary.
\end{lemma}

    By contradiction.
\end{proof}

\section{Expander Graphs}


\begin{definition}
  \label{def:expander_graph}
  \lean{ExpanderGraphs.IsExpander}
  \uses{def:cheegerConstant}
  \leanok
  A finite graph $G$ is a \textbf{$(k, \varepsilon)$-expander} if each vertex of $G$ has valency at most $k$, and $h(G) \ge \varepsilon > 0$.
\end{definition}

\begin{definition}
  \label{def:expander_sequence}
  \lean{ExpanderGraphs.expanderSequence}
  \uses{def:expander_graph}
  \leanok
  A sequence of finite graphs $\{G_i\}$ is called an \textbf{expander sequence} if $|G_i| \to \infty$ and there exists $k, \varepsilon$ such that each $G_i$ is a $(k, \varepsilon)$-expander.
\end{definition}
